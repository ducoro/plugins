真实视频合成命令（单视频 + 静止透明叠层 / 三连横条 / 四宫格）

# 1. 单视频 5s + 顶部静止透明标题层（文字不动，底下真实视频流畅流动）
python3 scripts/live_photo.py compose clip.mp4 -o drafts/<id>/assets/live-01.mov \
  --platform xiaohongshu --layout single --duration 5.0 --crop-x 0.16 \
  --overlay drafts/<id>/assets/overlay-top.png \
  --key-jpg drafts/<id>/assets/live-01.jpg --fallback-mp4 drafts/<id>/assets/live-01.mp4 --json

# 2. 三连横条 5s（16:9 横屏实拍/录屏首选，不裁坏主体）
python3 scripts/live_photo.py compose c1.mp4 c2.mp4 c3.mp4 -o drafts/<id>/assets/stack3.mov \
  --platform xiaohongshu --layout stack-3 --duration 5.0 --gap 8 \
  --key-jpg drafts/<id>/assets/stack3.jpg --fallback-mp4 drafts/<id>/assets/stack3.mp4 --json

# 3. 四宫格无字实况拼图 + macOS 原生打 .pvt 包并验收
python3 scripts/live_photo.py compose c1.mp4 c2.mp4 c3.mp4 c4.mp4 -o drafts/<id>/assets/quad.mov \
  --platform xiaohongshu --layout grid-4 --duration 5.0 --crop-x 0.12,0.16,0.72,0.18 --gap 8 \
  --key-jpg drafts/<id>/assets/quad.jpg --fallback-mp4 drafts/<id>/assets/quad.mp4 --json
python3 scripts/live_photo.py package-pvt drafts/<id>/assets/quad.jpg drafts/<id>/assets/quad.mov -o drafts/<id>/assets/quad.pvt --json
python3 scripts/live_photo.py validate drafts/<id>/assets/quad.pvt --platform xiaohongshu --json
